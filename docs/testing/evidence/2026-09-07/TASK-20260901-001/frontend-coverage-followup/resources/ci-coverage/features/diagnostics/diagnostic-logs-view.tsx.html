function contextText(row: DiagnosticLogRowViewModel): string {
  return [
    row.device_serial ? `设备 ${row.device_serial}` : null,
    row.listener_id ? `入口 ${row.listener_id}` : null,
    row.profile_id ? `方案 ${row.profile_id}` : null,
  ]
    .filter(Boolean)
    .join(" · ") || "—";
}
